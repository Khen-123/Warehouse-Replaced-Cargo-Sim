<script>
    const UI_UPDATE_INTERVAL = 500;
    let lastUpdate = 0;

    // Tab Switching Logic
    function switchTab(tab) {
        const tabs = ['stats', 'ship', 'up'];
        const btnMap = { 'stats': 'tab-stats', 'ship': 'tab-ship', 'up': 'tab-up' };
        const contentMap = { 'stats': 'content-stats', 'ship': 'content-ship', 'up': 'content-up' };

        tabs.forEach(t => {
            const btn = document.getElementById(btnMap[t]);
            const content = document.getElementById(contentMap[t]);
            
            if (t === tab) {
                btn.classList.add('bg-primary', 'text-on-primary', 'pixel-border-pressed', 'font-bold');
                btn.classList.remove('bg-surface-container', 'text-on-surface-variant', 'pixel-border');
                content.classList.remove('hidden');
            } else {
                btn.classList.remove('bg-primary', 'text-on-primary', 'pixel-border-pressed', 'font-bold');
                btn.classList.add('bg-surface-container', 'text-on-surface-variant', 'pixel-border');
                content.classList.add('hidden');
            }
        });
    }

    // Code Editor Syntax Highlighting
    const codeArea = document.getElementById('code-textarea');
    const highlightLayer = document.getElementById('code-editor-layer');

    function custom_syntax_highlighter(code) {
        // Escape HTML
        let html = code.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
        
        // Keywords
        const keywords = ['while', 'if', 'else', 'true', 'false'];
        keywords.forEach(kw => {
            const regex = new RegExp(`\\b${kw}\\b`, 'g');
            html = html.replace(regex, `<span class="token-keyword">${kw}</span>`);
        });

        // Functions/Actions
        const actions = ['move', 'pickup', 'dropoff', 'find', 'call', 'deliver', 'charge'];
        actions.forEach(action => {
            const regex = new RegExp(`\\b${action}\\b`, 'g');
            html = html.replace(regex, `<span class="token-function">${action}</span>`);
        });

        // Comments
        html = html.replace(/(\/\/.*)/g, '<span class="token-comment">$1</span>');

        // Strings/Params (simple check for uppercase constants)
        html = html.replace(/\b([A-Z_]+)\b/g, '<span class="token-string">$1</span>');

        return html;
    }

    function updateHighlighter() {
        highlightLayer.innerHTML = custom_syntax_highlighter(codeArea.value) + "\n";
    }

    codeArea.addEventListener('input', updateHighlighter);
    codeArea.addEventListener('scroll', () => {
        highlightLayer.scrollTop = codeArea.scrollTop;
    });

    // Initialize highlighter
    updateHighlighter();

    // Throttled UI Updates
    function updateDashboard(timestamp) {
        if (!lastUpdate || timestamp - lastUpdate >= UI_UPDATE_INTERVAL) {
            lastUpdate = timestamp;
            
            // Randomly fluctuate some stats for visual interest
            const throughput = (93 + Math.random() * 2).toFixed(1);
            const errors = (0.01 + Math.random() * 0.02).toFixed(3);
            const power = 80 + Math.floor(Math.random() * 10);
            
            const throughputEl = document.getElementById('ui-throughput');
            const throughputBar = document.getElementById('ui-throughput-bar');
            const errorsEl = document.getElementById('ui-errors');
            const powerEl = document.getElementById('ui-power');
            const graph = document.getElementById('flow-graph-container');

            if (throughputEl) throughputEl.textContent = `${throughput}%`;
            if (throughputBar) throughputBar.style.width = `${throughput}%`;
            if (errorsEl) errorsEl.textContent = `${errors}%`;
            if (powerEl) powerEl.textContent = `${power}%`;
            
            // Update flow graph
            if (graph) {
                const bars = graph.children;
                for (let bar of bars) {
                    bar.style.height = `${20 + Math.random() * 70}%`;
                }
            }
        }
        requestAnimationFrame(updateDashboard);
    }

    requestAnimationFrame(updateDashboard);
</script>
